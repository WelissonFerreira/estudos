rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /clientes/{clienteId}/pedidos/{pedidoId} {
      allow read, write: if true;
    }
    // Esta é uma regra padrão para outros documentos
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
